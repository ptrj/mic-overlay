cmake_minimum_required(VERSION 3.21)
project(mic-overlay VERSION 0.2.2 LANGUAGES CXX
    HOMEPAGE_URL "https://github.com/ptrj/mic-overlay")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick Qml Widgets Network)
find_package(Qt6 REQUIRED COMPONENTS Svg)
find_package(LayerShellQt 6.6 REQUIRED)
qt_standard_project_setup()

qt_add_library(mic-state STATIC src/micstatemonitor.cpp src/micstatemonitor.h
    src/pipewireevents.cpp src/pipewireevents.h)
target_link_libraries(mic-state PUBLIC Qt6::Core)
target_include_directories(mic-state PUBLIC src)
target_compile_options(mic-state PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(mic-overlay src/main.cpp src/shutdownhandler.cpp src/shutdownhandler.h)
qt_add_library(desktop-ui STATIC src/desktopui.cpp src/desktopui.h src/autostart.cpp src/autostart.h)
target_link_libraries(desktop-ui PUBLIC mic-state overlay-config Qt6::Widgets Qt6::Svg)
target_include_directories(desktop-ui PUBLIC src)
target_compile_options(desktop-ui PRIVATE -Wall -Wextra -Wpedantic)
qt_add_library(overlay-config STATIC src/overlayconfig.cpp src/overlayconfig.h)
target_link_libraries(overlay-config PUBLIC Qt6::Core)
target_include_directories(overlay-config PUBLIC src)
target_compile_options(overlay-config PRIVATE -Wall -Wextra -Wpedantic)
qt_add_library(overlay-controller STATIC src/overlaycontroller.cpp src/overlaycontroller.h)
target_link_libraries(overlay-controller PUBLIC Qt6::Gui LayerShellQt::Interface)
target_include_directories(overlay-controller PUBLIC src)
target_compile_options(overlay-controller PRIVATE -Wall -Wextra -Wpedantic)
qt_add_resources(mic-overlay "overlay" PREFIX "/" FILES
    qml/Main.qml assets/mic-on.svg assets/mic-muted.svg)
target_link_libraries(mic-overlay PRIVATE
    desktop-ui mic-state overlay-config overlay-controller Qt6::Network Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Svg LayerShellQt::Interface)
target_compile_definitions(mic-overlay PRIVATE MIC_OVERLAY_VERSION="${PROJECT_VERSION}")
target_compile_options(mic-overlay PRIVATE -Wall -Wextra -Wpedantic)

include(GNUInstallDirs)
install(TARGETS mic-overlay RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
set(MIC_OVERLAY_DESKTOP_EXECUTABLE "${CMAKE_INSTALL_FULL_BINDIR}/mic-overlay")
string(REPLACE "\\" "\\\\\\\\" MIC_OVERLAY_DESKTOP_EXECUTABLE "${MIC_OVERLAY_DESKTOP_EXECUTABLE}")
string(REPLACE "\"" "\\\\\"" MIC_OVERLAY_DESKTOP_EXECUTABLE "${MIC_OVERLAY_DESKTOP_EXECUTABLE}")
string(REPLACE "$" "\\\\$" MIC_OVERLAY_DESKTOP_EXECUTABLE "${MIC_OVERLAY_DESKTOP_EXECUTABLE}")
string(REPLACE "`" "\\\\`" MIC_OVERLAY_DESKTOP_EXECUTABLE "${MIC_OVERLAY_DESKTOP_EXECUTABLE}")
string(REPLACE "%" "%%" MIC_OVERLAY_DESKTOP_EXECUTABLE "${MIC_OVERLAY_DESKTOP_EXECUTABLE}")
configure_file(packaging/mic-overlay.desktop.in mic-overlay.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/mic-overlay.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES assets/mic-overlay.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
install(FILES config/config.ini.example DESTINATION "${CMAKE_INSTALL_DATADIR}/mic-overlay")
install(FILES docs/installation.md docs/configuration.md docs/packaging.md docs/acceptance.md
    DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/mic-overlay")

    set_tests_properties(mic-state PROPERTIES TIMEOUT 60)
    qt_add_executable(pipewire-events-test tests/pipewireevents_test.cpp)

    add_test(NAME pipewire-events COMMAND pipewire-events-test)

    target_link_libraries(mic-state-probe PRIVATE mic-state)
    qt_add_executable(overlay-config-test tests/overlayconfig_test.cpp)
    target_link_libraries(overlay-config-test PRIVATE overlay-config Qt6::Test)

    qt_add_executable(tray-gui-smoke tests/tray_gui_smoke.cpp src/shutdownhandler.cpp)

include(cmake/Packaging.cmake)
